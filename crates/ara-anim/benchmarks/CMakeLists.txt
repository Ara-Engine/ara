cmake_minimum_required(VERSION 3.30)
project(ara_ozz_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(ozz_build_tools OFF CACHE BOOL "" FORCE)
set(ozz_build_fbx OFF CACHE BOOL "" FORCE)
set(ozz_build_gltf OFF CACHE BOOL "" FORCE)
set(ozz_build_samples OFF CACHE BOOL "" FORCE)
set(ozz_build_howtos OFF CACHE BOOL "" FORCE)
set(ozz_build_tests OFF CACHE BOOL "" FORCE)
set(ozz_build_msvc_rt_dll ON CACHE BOOL "" FORCE)
add_subdirectory(../perf-local/ozz ozz-build)
add_executable(ozz_bench ozz_bench.cpp ../perf-local/ozz/extern/jsoncpp/dist/jsoncpp.cpp)
target_include_directories(ozz_bench PRIVATE ../perf-local/ozz/extern/jsoncpp/dist)
target_link_libraries(ozz_bench PRIVATE ozz_animation_offline)
if(MSVC)
  target_compile_options(ozz_bench PRIVATE /O2 /fp:precise)
endif()
